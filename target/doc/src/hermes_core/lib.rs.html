28#[cfg(test)]
29mod tests {
30    use super::config::*;
31    use std::{fs, path::Path};
32    use tempfile::tempdir;
33
34    #[test]
35    fn resolves_explicit_existing_home() {
36        let d = tempdir().unwrap();
37        assert_eq!(resolve_hermes_home(Some(d.path())).unwrap(), d.path());
38    }
39    #[test]
40    fn returns_error_when_home_missing() {
41        let e =
42            resolve_hermes_home(Some(Path::new("/definitely/missing/hermes-home"))).unwrap_err();
43        assert!(matches!(e, ConfigError::HomeNotFound { .. }));
44    }
45    #[test]
46    fn parses_valid_hermes_yaml_read_only() {
47        let d = tempdir().unwrap();
48        fs::write(d.path().join("config.yaml"), "model:\n  default: gpt-test\n  provider: custom\n  base_url: http://localhost:8080/v1\n  api_key: secret-value\n").unwrap();
49        let c = load_config(d.path()).unwrap();
50        assert_eq!(c.model.default.as_deref(), Some("gpt-test"));
51        assert_eq!(c.model.provider.as_deref(), Some("custom"));
52        assert_eq!(c.model.api_key.as_ref().unwrap().expose(), "secret-value");
53        assert_eq!(format!("{:?}", c.model), "ModelConfig { default: Some(\"gpt-test\"), provider: Some(\"custom\"), base_url: Some(\"http://localhost:8080/v1\"), api_key: Some(SecretString(***REDACTED***)) }");
54        assert!(!d.path().join("config.yaml.tmp").exists());
55    }
56    #[test]
57    fn applies_explicit_provider_override_without_writing() {
58        let d = tempdir().unwrap();
59        let path = d.path().join("config.yaml");
60        fs::write(&path, "model:\n  provider: auto\n").unwrap();
61        let c = load_config_with_provider(d.path(), Some("openai")).unwrap();
62        assert_eq!(c.model.provider.as_deref(), Some("openai"));
63        assert_eq!(
64            fs::read_to_string(path).unwrap(),
65            "model:\n  provider: auto\n"
66        );
67    }
68
69    #[test]
70    fn rejects_invalid_yaml() {
71        let d = tempdir().unwrap();
72        fs::write(d.path().join("config.yaml"), "model: [broken").unwrap();
73        assert!(matches!(
74            load_config(d.path()),
75            Err(ConfigError::ParseFailed { .. })
76        ));
77    }
78    #[test]
79    fn reports_missing_config() {
80        let d = tempdir().unwrap();
81        assert!(matches!(
82            load_config(d.path()),
83            Err(ConfigError::ConfigFileMissing { .. })
84        ));
85    }
